#+begin_src python
def add_implicit_sink(
    lhs: Dict[str, fp.ImplicitSourceTerm],
    rates: Dict[str, float],
    species: str,
    coeff: Coeff,
    rate: Rate,
) -> fp.ImplicitSourceTerm:
    """
    Add a *linear* sink  – ‑coeff·X – to the equation for ``species``.
    The term is represented by an :class:`~fipy.terms.implicitSourceTerm.ImplicitSourceTerm`.

    Parameters
    ----------
    lhs
        Dictionary that stores the (already created) ``ImplicitSourceTerm`` objects
        for each species.  If the entry does not exist yet a new term is created.
    rates
        Dictionary that accumulates the *raw* (non‑linear) reaction rates for reporting.
    species
        Name of the variable that receives the sink.
    coeff
        Linear coefficient that multiplies the variable (‑coeff is the Jacobian entry).
    rate
        The *non‑linear* part of the rate (used only for bookkeeping).

    Returns
    -------
    ImplicitSourceTerm
        The term that has been added / updated.
    """
    # -----------------------------------------------------------------
    # 1️⃣  Make sure we have a term object for this species
    # -----------------------------------------------------------------
    if species not in lhs:
        # ``var`` will be filled later by the caller – we just need a placeholder.
        lhs[species] = fp.ImplicitSourceTerm(var=None, coeff=0.0)

    term = lhs[species]

    # -----------------------------------------------------------------
    # 2️⃣  Update the linear coefficient (the Jacobian entry)
    # -----------------------------------------------------------------
    # FiPy stores the coefficient as a *positive* number that is multiplied
    # by the variable on the LHS.  Because we want a sink we add a *negative*
    # contribution.
    term.coeff = term.coeff - float(coeff)   # coeff may be an array → cast to float for safety

    # -----------------------------------------------------------------
    # 3️⃣  Store the raw (non‑linear) rate for diagnostics
    # -----------------------------------------------------------------
    rates[species] = rates.get(species, 0.0) + float(np.asarray(rate).sum())

    return term


def add_implicit_coupling(
    cross: Dict[str, list[Tuple[str, Coeff]]],
    rates: Dict[str, float],
    target_species: str,
    source_species: str,
    coeff: Coeff,
    rate: Rate,
) -> None:
    """
    Add a *linear* source that couples ``source_species`` to ``target_species``:

    .. math::
        \\frac{d\\,target}{dt} = +\\,coeff \\; source .

    The coupling is stored in ``cross`` so that, after the main solve, the
    caller can add the corresponding ``ImplicitSourceTerm`` objects to the
    *source* equations (see the main routine below).

    Parameters
    ----------
    cross
        ``{target: [(source, coeff), …]}`` – a bookkeeping container used by the
        outer solver to build the full Jacobian.
    rates
        Dictionary that accumulates the raw reaction rates (for reporting).
    target_species
        Name of the variable that receives the source.
    source_species
        Name of the variable that supplies the source.
    coeff
        Linear coefficient that multiplies ``source_species``.
    rate
        The non‑linear rate (only stored for diagnostics).
    """
    cross.setdefault(target_species, []).append((source_species, float(coeff)))
    rates[target_species] = rates.get(target_species, 0.0) + float(np.asarray(rate).sum())

If you would like to *force the system onto the equilibrium manifold after each solve*
(e.g. when the equilibrium constant is astronomically large), use the tiny helper
below.  It works on any FiPy =CellVariable= that participates in the Fe‑S pair.

#+begin_src python
def project_fe_s_equilibrium(
    fe2: Var,
    hs: Var,
    fes: Var,
    K_eq: float,
    porosity: float = 1.0,
) -> None:
    """
    Enforce the algebraic relation   K_eq = [FeS] / ([Fe2][H2S])
    by projecting the three fields onto the closest point that satisfies it.
    The projection conserves the total amount of Fe + S (mass balance).

    The algorithm solves, cell‑wise, a small 2‑by‑2 linear system for the
    corrected dissolved concentrations; the solid mass is then updated to keep
    the total Fe (or S) constant.

    Parameters
    ----------
    fe2, hs, fes
        FiPy ``CellVariable`` objects for dissolved Fe²⁺, dissolved H₂S and the
        solid FeS pool (treated as a mass per unit volume).
    K_eq
        Equilibrium constant (dimensionless).  Typical values for FeS are
        > 10⁶, so the reaction is essentially instantaneous.
    porosity
        Fraction of the cell volume that is pore space – used to convert solid
        mass to a concentration that participates in the equilibrium.
    """
    # 1️⃣  Convert solid mass to an *equivalent* dissolved concentration
    solid_eq = fes / porosity

    # 2️⃣  Solve for the corrected dissolved concentrations.
    #    We keep the *total* Fe (dissolved + solid) constant.
    total_fe = fe2 + solid_eq
    #    The equilibrium condition reads  fe2_corr * hs_corr = solid_eq_corr / K_eq
    #    Together with  fe2_corr + solid_eq_corr = total_fe  we obtain a quadratic.
    a = 1.0
    b = -total_fe
    c = solid_eq / K_eq
    disc = b ** 2 - 4 * a * c
    # Guard against negative discriminant due to round‑off.
    disc = np.where(disc < 0, 0.0, disc)

    fe2_corr = (-b - np.sqrt(disc)) / (2 * a)          # the physically realistic root
    hs_corr = (solid_eq / K_eq) / fe2_corr

    # 3️⃣  Update the fields *in place*.
    fe2.setValue(fe2_corr)
    hs.setValue(hs_corr)
    #    The solid mass follows from the mass balance:
    fes.setValue(porosity * (total_fe - fe2_corr))
#+end_src

#+begin_src python
def fes_formation(
    c: Mapping[str, Var],               # concentration dictionary (fe2_total, ts2, fes, …)
    k: Mapping[str, float],             # kinetic constants (fes_isp, fes_isd, fes_sp, hplus, …)
    lim: Mapping[str, float],           # limiting‑factor scalars (e.g. "fes_explicit")
    mp: Mapping[str, float],            # model parameters (fac_s, f_diss, dt_max, …)
    LHS: Dict[str, fp.ImplicitSourceTerm],
    RHS: Dict[str, fp.ExplicitSourceTerm],   # you already have a similar dict for explicit terms
    RATES: Dict[str, float],
    CROSS: Dict[str, list[Tuple[str, Coeff]]],
) -> None:
    """
    Assemble the FiPy terms for the Fe‑S precipitation / dissolution system.

    The function follows the *implicit‑fast‑chemistry* strategy:
    * precipitation (Ω>1) → implicit sink on Fe²⁺ and H₂S, implicit source on FeS
    * dissolution (Ω≤1) → implicit source on Fe²⁺ and H₂S, implicit sink on FeS
    * a tiny explicit term (`add_explicit_source`) is kept for the *slow* part of
      the precipitation rate (the ``k_isp_eff`` factor after the limiter).

    All matrix entries are added via the two helper functions above, so the
    surrounding code does not have to be changed.
    """
    # -----------------------------------------------------------------
    # 0️⃣  Grab the FiPy variables (they are stored in the ``c`` dict)
    # -----------------------------------------------------------------
    fe2: Var = c["fe2_total"]
    hs:  Var = c["ts2"]
    fes: Var = c["fes"]

    # -----------------------------------------------------------------
    # 1️⃣  Current values (as plain NumPy arrays) – needed for the *non‑linear*
    #      coefficients that go into the Jacobian.
    # -----------------------------------------------------------------
    fe2_val   = fe2.value
    fe2_liq   = fe2_val * mp["f_diss"]               # dissolved fraction only
    hs_val    = hs.value
    fes_val   = fes.value

    # -----------------------------------------------------------------
    # 2️⃣  Saturation state Ω = [Fe2][H2S] / ( [H⁺]·K_sp )
    # -----------------------------------------------------------------
    omega_den = k["hplus"] * k["fes_sp"] + 1e-30
    omega     = (fe2_liq * hs_val) / omega_den

    # Boolean masks (as float) – they allow us to write *one* expression that works
    # for the whole mesh.
    is_precip = (omega > 1.0).astype(float)   # 1 where Ω>1, else 0
    is_diss   = 1.0 - is_precip                # complementary mask

    # -----------------------------------------------------------------
    # 3️⃣  Precipitation (fast) – implicit part
    # -----------------------------------------------------------------
    # Raw (linearised) rate: k_isp * Ω
    # We later limit it so that we never consume >90 % of a reactant in one dt.
    raw_rate = k["fes_isp"] * (omega - 1.0) * is_precip
    raw_rate = np.maximum(raw_rate, 0.0)

    # -----------------------------------------------------------------
    # 4️⃣  Limiter – keep the reaction from draining a reactant too fast
    # -----------------------------------------------------------------
    dt_sim = mp.get("dt_max", 1.0)   # fallback for steady‑state runs
    max_rate_fe2 = (0.9 * fe2_liq) / (dt_sim + 1e-30)
    max_rate_hs  = (0.9 * hs_val)  / (dt_sim + 1e-30)
    max_allowed  = np.minimum(max_rate_fe2, max_rate_hs)

    limiter = np.minimum(1.0, max_allowed / (raw_rate + 1e-30))
    k_isp_eff = k["fes_isp"] * limiter    # *effective* kinetic constant after limiting

    # -----------------------------------------------------------------
    # 5️⃣  Coefficients that appear in the Jacobian
    # -----------------------------------------------------------------
    # Precipitation term that multiplies the *dissolved* concentrations:
    #   -k_eff * H2S / ω_den   for Fe2
    #   -k_eff * Fe2_liq / ω_den   for H2S
    common_precip = k_isp_eff * is_precip / omega_den

    coeff_fe2_sink = common_precip * hs_val * mp["f_diss"]           # ∂R/∂Fe2
    coeff_hs_sink  = common_precip * fe2_liq                     # ∂R/∂H2S
    coeff_fes_src  = common_precip * hs_val * mp["f_diss"] * mp["fac_s"]   # ∂R/∂FeS (source)

    # -----------------------------------------------------------------
    # 6️⃣  Dissolution (slow) – implicit part
    # -----------------------------------------------------------------
    # Ω_diss = min(Ω, 1)  → (1‑Ω_diss) is the driving force
    omega_diss   = np.minimum(omega, 1.0)
    fes_diss_coeff = k["fes_isd"] * (1.0 - omega_diss) * is_diss * lim["fes_explicit"]
    # This coefficient multiplies the solid concentration (FeS) and appears as:
    #   +k_isd*(1‑Ω)   for Fe2   (source)
    #   +k_isd*(1‑Ω)   for H2S   (source)
    #   -k_isd*(1‑Ω)   for FeS   (sink)

    # -----------------------------------------------------------------
    # 7️⃣  Assemble the implicit terms via the helpers
    # -----------------------------------------------------------------
    # ---- Fe2 (liquid) ------------------------------------------------
    add_implicit_sink(LHS, RATES, "fe2_total", coeff_fe2_sink,
                      coeff_fe2_sink * fe2_val)                # precipitation sink
    add_implicit_coupling(CROSS, RATES, "fe2_total", "fes",
                          fes_diss_coeff,
                          fes_diss_coeff * fes_val)               # dissolution source

    # ---- H2S ---------------------------------------------------------
    add_implicit_sink(LHS, RATES, "ts2", coeff_hs_sink,
                      coeff_hs_sink * hs_val)                    # precipitation sink
    # Explicit part of the *fast* precipitation (the “k_isp_eff * is_precip”
    # factor that is not linearised) – kept as an explicit source so that the
    # overall rate stays exactly k_isp*(Ω‑1) after the limiter.
    RHS.setdefault("ts2", fp.ExplicitSourceTerm(var=hs, value=0.0))
    RHS["ts2"].value += k_isp_eff * is_precip

    add_implicit_coupling(CROSS, RATES, "ts2", "fes",
                          fes_diss_coeff,
                          fes_diss_coeff * fes_val)               # dissolution source

    # ---- FeS (solid) -------------------------------------------------
    # Precipitation source (coupled to Fe2)
    add_implicit_coupling(CROSS, RATES, "fes", "fe2_total",
                          coeff_fes_src,
                          coeff_fes_src * fe2_val)

    # Explicit correction for the fast precipitation (the “‑k_isp_eff*is_precip*fac_s” term)
    RHS.setdefault("fes", fp.ExplicitSourceTerm(var=fes, value=0.0))
    RHS["fes"].value -= k_isp_eff * is_precip * mp["fac_s"]

    # Dissolution sink on the solid
    coeff_fes_sink = fes_diss_coeff * mp["fac_s"]
    add_implicit_sink(LHS, RATES, "fes", coeff_fes_sink,
                      coeff_fes_sink * fes_val)

    # -----------------------------------------------------------------
    # 8️⃣  OPTIONAL: isotope handling (32S) – unchanged apart from the
    #            helper calls (they still work with the new LHS dict)
    # -----------------------------------------------------------------
    if "ts2_32" in c:
        ts2_32: Var = c["ts2_32"]
        fes_32: Var = c["fes_32"]

        ts2_inv = 1.0 / (hs_val + 1e-20)

        coeff_ts2_32_precip = (
            k_isp_eff * (fe2_liq / omega_den - ts2_inv) * is_precip
        )
        coeff_ts2_32_precip = np.maximum(coeff_ts2_32_precip, 0.0)

        # sink on H2S_32
        add_implicit_sink(
            LHS,
            RATES,
            "ts2_32",
            coeff_ts2_32_precip,
            coeff_ts2_32_precip * ts2_32.value,
        )

        # source on FeS_32 (coupled)
        add_implicit_coupling(
            CROSS,
            RATES,
            "fes_32",
            "ts2_32",
            coeff_ts2_32_precip * mp["fac_s"],
            coeff_ts2_32_precip * ts2_32.value * mp["fac_s"],
        )

        # dissolution sink on FeS_32
        add_implicit_sink(
            LHS, RATES, "fes_32", coeff_fes_sink, coeff_fes_sink * fes_32.value
        )

        # source on H2S_32 (coupled)
        add_implicit_coupling(
            CROSS,
            RATES,
            "ts2_32",
            "fes_32",
            fes_diss_coeff,
            fes_diss_coeff * fes_32.value,
        )
#+end_src

*** Short answer  

The /90 % limiter/ is *only a safety net* that prevents the explicit part of the fast precipitation from driving a concentration negative in a single time‑step.  
It does *not* represent a physical kinetic constraint, so it can indeed make the fast reaction appear slower and give a competing (slower) reaction a chance to “win”.  

If you need the chemistry to be /truly/ fast you have three robust alternatives that avoid that artefact:

| What you do now (limiter) | Recommended alternatives |
|---------------------------|--------------------------|
| Clip the rate so that no more than 90 % of a reactant is removed in one Δt. | *(A) Fully implicit treatment* – put the whole precipitation term into an =ImplicitSourceTerm=. FiPy’s Newton solver will handle the stiffness, so no limiter is needed. |
| Result: the fast reaction is artificially throttled and can be out‑competed by a slower kinetic term. | *(B) Algebraic equilibrium (projection)* – treat the Fe²⁺ + H₂S ⇌ FeS as an /equilibrium/ (K≫1). After each solve call =project_fe_s_equilibrium(...)= to force the system onto the equilibrium manifold. This is mathematically equivalent to an infinitely fast reaction. |
| | *(C) Sub‑cycling / adaptive Δt* – keep the current implicit + explicit split but automatically shrink Δt whenever =raw_rate*Δt > 0.9*conc=. The solver will take many tiny sub‑steps for the fast chemistry while the transport step stays large. |

Below I explain *why* the limiter can be problematic, and then give concrete code snippets for *A* and *B* (the two cleanest ways to get rid of the artefact).  

---

** 1. Why the 90 % limiter can change the chemistry

1. *Numerical vs. physical restriction* – The limiter is introduced only to keep the explicit part from producing a negative concentration. It does *not* appear in the governing PDEs, so the “effective” kinetic constant becomes  
   \[
   k_{\text{eff}} = k_{\text{orig}} \times \min\!\Bigl(1,\; \frac{0.9\,C}{k_{\text{orig}}\,\Omega\,\Delta t}\Bigr)
   \]
   which can be orders of magnitude smaller than the true k when Δt is large.

2. *Competing reactions see a reduced source term* – Suppose you have a second reaction with a true rate constant \(k_2\) of order 1–8 s⁻¹. If the fast precipitation is limited to, say, 0.01 s⁻¹ because Δt is too big, the second reaction will dominate even though physically it should be negligible.

3. *Mass‑balance is still OK, but the /pathway/ is wrong* – The limiter preserves total Fe and S, but the /order/ in which they are transferred between dissolved and solid pools is altered, which can affect downstream species (e.g., Fe³⁺ formed by oxidation of Fe²⁺).

In short: *if you care about the competition between a fast equilibrium and a slower kinetic step, you must remove the limiter*.

def fes_formation_fully_implicit(
    c: Mapping[str, fp.CellVariable],
    k: Mapping[str, float],
    lim: Mapping[str, float],
    LHS: Dict[str, fp.ImplicitSourceTerm],
    RHS: Dict[str, fp.ExplicitSourceTerm],
    RATES: Dict[str, float],
    CROSS: Dict[str, List[Tuple[str, float]]],
    mp: Mapping[str, float],
) -> None:

    Fast FeS precipitation is treated *entirely* implicitly.
    The explicit residual is omitted because it is the source of the
    ``9e+36`` blow‑up.
    """
    fe2 = c["fe2_total"]
    hs = c["ts2"]
    fes = c["fes"]

    # ----- 0. old‑step values (used only for the Jacobian) -----
    fe2_old = fe2.value * mp["f_diss"]
    hs_old = hs.value
    fes_old = fes.value

    # ----- 1. saturation index Ω -----
    omega_den = k["hplus"] * k["fes_sp"] + 1e-30
    omega = (fe2_old * hs_old) / omega_den

    precip_mask = (omega > 1.0).astype(float)  # 1 where Ω>1, else 0
    diss_mask = 1.0 - precip_mask

    # ----- 2. Implicit precipitation coefficients (Jacobian) -----
    #   R = k_isp * (Ω‑1)   →   ∂R/∂Fe2 = k_isp * hs / ω_den
    coeff_fe2 = -k["fes_isp"] * hs_old / omega_den * mp["f_diss"] * precip_mask
    coeff_hs = -k["fes_isp"] * fe2_old / omega_den * precip_mask
    coeff_fes = k["fes_isp"] * mp["fac_s"] * precip_mask  # source on solid

    # ----- 3. Add the implicit terms via the helper utilities -----
    add_implicit_sink(
        LHS, RATES, "fe2_total", coeff_fe2, coeff_fe2 * fe2.value
    )  # sink on Fe2
    add_implicit_sink(LHS, RATES, "ts2", coeff_hs, coeff_hs * hs.value)  # sink on H2S
    add_implicit_coupling(
        CROSS, RATES, "fes", "fe2_total", coeff_fes, coeff_fes * fe2.value

        CROSS, RATES, "fe2_total", "fes", fes_diss_coeff, fes_diss_coeff * fes.value

        CROSS, RATES, "ts2", "fes", fes_diss_coeff, fes_diss_coeff * fes.value
    )

    add_implicit_sink(LHS, RATES, "fes", coeff_fes_sink, coeff_fes_sink * fes.value)

        fes_32 = c["fes_32"]

**Why the fully‑implicit version blows up*  

Below is the chain of things that happen when you call  

#+begin_src python
eq = (TransientTerm() == …)      # built from the LHS / RHS dictionaries
eq.solve(dt=31536000.)           # one‑year time step
#+end_src

| Step | What FiPy does | What the code you posted does | Why it becomes unstable |
|------|----------------|-----------------------------|------------------------|
| *1. Build the Jacobian* | FiPy creates a sparse matrix *A* = /I/ + Δt · J, where *J* is the Jacobian of all implicit source terms. | You linearise the precipitation rate *R = kₚ (Ω‑1)* around the /old/ concentrations (=fe2_old=, =hs_old=). The Jacobian entries you feed to =add_implicit_sink= / =add_implicit_coupling= are therefore *constant* during the Newton iteration. | The true Jacobian of *R* depends on the /new/ concentrations (Ω contains the unknowns). By fixing it to the old values you break the Newton consistency – the residual is a /non‑linear/ function but the matrix you solve with is the /linearisation at the previous step/. When Δt is large the mismatch is huge and the Newton step cannot reduce the residual. |
| *2. Mask‑based “on/off” switch* | The Jacobian is a smooth function of the solution; the Newton method can follow the curvature. | You introduce =precip_mask = (omega > 1.0).astype(float)= (and =diss_mask = 1‑precip_mask=). This is a *discontinuous Heaviside* that flips from 0 to 1 when Ω crosses 1.0. The matrix you assemble can change /abruptly/ from one Newton iteration to the next. | The Newton algorithm assumes the matrix varies /smoothly/ with the unknowns. A step‑function Jacobian makes the linear system ill‑conditioned (often singular) the moment a cell switches from “no precipitation” to “precipitation”. FiPy therefore reports /“matrix is singular / solver failed”/ or simply diverges. |
| *3. Large Δt multiplies the Jacobian* | Implicit terms are multiplied by Δt inside FiPy (=A = I + Δt·J=). | Your precipitation coefficient is =k_isp ≈ 1.1e‑4 mol m⁻³ s⁻¹=. With Δt = 3.15 × 10⁷ s the effective diagonal contribution is roughly =Δt·k_isp ≈ 3 × 10³=. The sign you give is *negative* (=coeff_fe2= and =coeff_hs= are negative because you call =add_implicit_sink=). | A matrix with a *large negative diagonal* and comparable off‑diagonal entries loses diagonal dominance. The linear solver (usually a sparse LU or GMRES) becomes unstable, the condition number explodes, and the Newton iteration cannot converge. |
| *4. Mass‑balance mismatch* | Implicit source terms automatically keep the total mass consistent because the same coefficient appears with opposite sign on the coupled equations. | You add a sink on =fe2_total= /and/ a sink on =ts2=, but you *do not* add the /corresponding/ source term on =fes= with the /same/ coefficient; you add a /different/ coefficient (=coeff_fes = k_isp·fac_s·mask=). If =fac_s ≠ 1= the total Fe+S is not conserved in the linearised system. | The residual contains a hidden source/sink that the linear solver cannot satisfy → divergence. (Even if =fac_s = 1=, the sign error described in point 3 still makes the matrix singular.) |

In short, the failure is *not* a bug in FiPy; it is a consequence of trying to treat a /strongly non‑linear, discontinuous/ reaction as a /purely linear implicit source/ while using a very large time step.

*** 1. Keep the reaction /implicit/ *but linearise around the /new/ iterate*

    LHS: Dict[str, fp.ImplicitSourceTerm],

    fes: fp.CellVariable,

    hplus: float,

       R = k_isp * ( (fe2*f_diss*hs)/(hplus*fes_sp) - 1 )   (Ω‑1)
       The term is added to the three equations with the correct sign.

    # We therefore build three separate ImplicitSourceTerm objects,
    # each one knows its own variable.

    RATES["ts2"]       = RATES.get("ts2", 0.0)       + float(rate.mean())

def fes_formation_fully_implicit(
    c: Mapping[str, fp.CellVariable],
    k: Mapping[str, float],
    lim: Mapping[str, float],
    LHS: Dict[str, fp.ImplicitSourceTerm],
    RHS: Dict[str, fp.ExplicitSourceTerm],
    RATES: Dict[str, float],
    CROSS: Dict[str, List[Tuple[str, float]]],
    mp: Mapping[str, float],
) -> None:
    """
    Assemble the FiPy matrices for the Fe‑S system with **all fast
    precipitation terms treated implicitly**.

    The implementation follows the same public API that the rest of the
    code expects:

    * ``add_implicit_sink``   – adds a linear sink (‑coeff·var) to a species.
    * ``add_implicit_coupling`` – adds a linear source that couples one
      variable to another.

    The *only* change is that the precipitation reaction
    ``Fe²⁺ + H₂S ⇌ FeS`` is written as a *single* ``ImplicitSourceTerm``
    that contains the full non‑linear expression
    ``R = k_isp·(Ω‑1)`` where

        Ω = (Fe²⁺·f_diss·H₂S) / (H⁺·K_sp)

    Because the term is fully implicit the Newton solver sees the
    correct Jacobian; no artificial “90 % limiter” or old‑step linearisation
    is required, and the large one‑year time step (dt≈3.15e7 s) remains stable.

    Parameters
    ----------
    c, k, lim, mp
        Dictionaries that hold the FiPy variables, kinetic constants,
        user‑defined limiting factors and model parameters (porosity,
        dissolution fraction, etc.).  The keys used here are exactly the
        same as in the original routine.
    LHS, RHS, RATES, CROSS
        Containers that collect the implicit/explicit source terms,
        diagnostic rate totals and the coupling information required
        later when the global FiPy equation is built.
    """
    # -----------------------------------------------------------------
    # 0️⃣  Grab the FiPy cell‑variables
    # -----------------------------------------------------------------
    fe2: fp.CellVariable = c["fe2_total"]
    hs:  fp.CellVariable = c["ts2"]
    fes: fp.CellVariable = c["fes"]

    # -----------------------------------------------------------------
    # 1️⃣  Build the *non‑linear* precipitation rate
    # -----------------------------------------------------------------
    # Saturation index Ω (vector‑valued, depends on the unknowns)
    omega_den = k["hplus"] * k["fes_sp"] + 1e-30
    omega = (fe2 * mp["f_diss"] * hs) / omega_den          # FiPy expression

    # Full (backward‑Euler) rate:  R = k_isp·(Ω‑1)
    rate_precip = k["fes_isp"] * (omega - 1.0)             # FiPy CellVariable

    # -----------------------------------------------------------------
    # 2️⃣  Implicit source terms for the three species
    # -----------------------------------------------------------------
    #   Fe²⁺  :  -R·f_diss   (sink)
    #   H₂S   :  -R          (sink)
    #   FeS   :  +R·fac_s    (source)
    #
    #  We create three independent ImplicitSourceTerm objects and store
    #  them in the ``LHS`` dictionary.  If a term already exists for the
    #  species we simply add the new coefficient to the existing one
    #  (FiPy allows ``term.coeff += …``).
    #
    #  NOTE:  The helper functions are *not* used for these three lines
    #         because they expect a *linear* coefficient.  The rest of
    #         the routine (slow dissolution, isotopes) continues to use
    #         the original helpers unchanged.

    # ---- Fe²⁺ -------------------------------------------------------
    term_fe2 = fp.ImplicitSourceTerm(var=fe2,
                                    coeff=-rate_precip * mp["f_diss"])
    if "fe2_total" in LHS:
        LHS["fe2_total"].coeff += term_fe2.coeff
    else:
        LHS["fe2_total"] = term_fe2

    # ---- H₂S --------------------------------------------------------
    term_hs = fp.ImplicitSourceTerm(var=hs,
                                    coeff=-rate_precip)
    if "ts2" in LHS:
        LHS["ts2"].coeff += term_hs.coeff
    else:
        LHS["ts2"] = term_hs

    # ---- FeS (solid) -------------------------------------------------
    term_fes = fp.ImplicitSourceTerm(var=fes,
                                     coeff=rate_precip * mp["fac_s"])
    if "fes" in LHS:
        LHS["fes"].coeff += term_fes.coeff
    else:
        LHS["fes"] = term_fes

    # -----------------------------------------------------------------
    # 3️⃣  Book‑keeping of the (non‑linear) precipitation rate
    # -----------------------------------------------------------------
    # ``RATES`` is only for diagnostics – we store the *cell‑averaged* value.
    avg_rate = float(rate_precip.mean())
    RATES["fe2_total"] = RATES.get("fe2_total", 0.0) + avg_rate
    RATES["ts2"]       = RATES.get("ts2", 0.0)       + avg_rate
    RATES["fes"]       = RATES.get("fes", 0.0)       + avg_rate

    # -----------------------------------------------------------------
    # 4️⃣  Slow dissolution (still linear, therefore we keep the helpers)
    # -----------------------------------------------------------------
    #   Ω_diss = min(Ω, 1)   →  (1‑Ω_diss) is the driving force
    omega_diss = fp.minimum(omega, 1.0)                     # FiPy expression
    diss_mask  = 1.0 - fp.where(omega > 1.0, 1.0, 0.0)      # same as 1‑precip_mask

    fes_diss_coeff = (
        k["fes_isd"] * (1.0 - omega_diss) * diss_mask * lim["fes_explicit"]
    )   # this is a *linear* coefficient that multiplies the solid

    # Fe²⁺ source from dissolution
    add_implicit_coupling(
        CROSS, RATES,
        target_species="fe2_total",
        source_species="fes",
        coeff=fes_diss_coeff,
        rate=fes_diss_coeff * fes.value,
    )

    # H₂S source from dissolution
    add_implicit_coupling(
        CROSS, RATES,
        target_species="ts2",
        source_species="fes",
        coeff=fes_diss_coeff,
        rate=fes_diss_coeff * fes.value,
    )

    # FeS sink from dissolution (multiply by the solid‑phase factor)
    coeff_fes_sink = fes_diss_coeff * mp["fac_s"]
    add_implicit_sink(
        LHS, RATES, "fes", coeff_fes_sink, coeff_fes_sink * fes.value
    )

    # -----------------------------------------------------------------
    # 5️⃣  Optional isotope bookkeeping (unchanged – still uses helpers)
    # -----------------------------------------------------------------
    if "ts2_32" in c:
        ts2_32: fp.CellVariable = c["ts2_32"]
        fes_32: fp.CellVariable = c["fes_32"]

        # 5a – precipitation (non‑linear part) for 32S
        ts2_inv = 1.0 / (hs.value + 1e-20)

        coeff_ts2_32_precip = (
            k["fes_isp"] *
            (fe2 * mp["f_diss"] / omega_den - ts2_inv) *
            fp.where(omega > 1.0, 1.0, 0.0)
        )
        coeff_ts2_32_precip = fp.maximum(coeff_ts2_32_precip, 0.0)

        add_implicit_sink(
            LHS,
            RATES,
            "ts2_32",
            coeff_ts2_32_precip,
            coeff_ts2_32_precip * ts2_32.value,
        )

        # 5b – coupling of 32S to the solid
        add_implicit_coupling(
            CROSS,
            RATES,
            target_species="fes_32",
            source_species="ts2_32",
            coeff=coeff_ts2_32_precip * mp["fac_s"],
            rate=coeff_ts2_32_precip * ts2_32.value * mp["fac_s"],
        )

        # 5c – dissolution sink on the solid (same coeff as for the main solid)
        add_implicit_sink(
            LHS,
            RATES,
            "fes_32",
            coeff_fes_sink,
            coeff_fes_sink * fes_32.value,
        )

        # 5d – dissolution source on H₂S_32 (coupled to solid)
        add_implicit_coupling(
            CROSS,
            RATES,
            target_species="ts2_32",
            source_species="fes_32",
            coeff=fes_diss_coeff,
            rate=fes_diss_coeff * fes_32.value,
        )
#+end_src

| Old version | New version |
|------------|-------------|
| Linearised coefficients were computed with *old‑step* concentrations (=fe2_old=, =hs_old=). | The precipitation rate is built *directly* from the FiPy variables (=fe2=, =hs=) – FiPy evaluates the Jacobian at the current Newton iterate. |

** 1.  What the original helper does

#+begin_src python
def add_implicit_sink(LHS, RATES, species, coeff, rate):
    # LHS[species] is a *scalar/array* that represents the diagonal entry
    # of the Jacobian for this species.
    LHS[species] = LHS[species] - coeff          # ← subtract because it is a sink
    RATES[species] = RATES.get(species, 0.0) - float(np.asarray(rate).sum())
#+end_src

- =coeff= is *the Jacobian entry* (‑∂R/∂[Fe²⁺] in our case).  
- =rate= is only for diagnostics; it is *not* used by the linear solver.

When the time‑loop later builds the global FiPy equation it does something like:

#+begin_src python
for species_name, var in zip(species_names, variables):
    lhs_coeff = lhs_vars[species_name]                 # the number we stored above
    lhs_term  = fp.ImplicitSourceTerm(coeff=lhs_coeff, var=var)
    # rhs_term is whatever explicit source you added elsewhere
    equation = (fp.TransientTerm(var=var) == lhs_term + rhs_term)

** 3.  Which style should you keep?

| Situation | Recommended style |
|-----------|-------------------|
| *You want to keep the original helper‑based code untouched* (e.g. many other modules already call =add_implicit_sink= and expect =LHS[species]= to be a numeric coefficient). | Keep =add_implicit_sink= *as‑is* and /store only the coefficient/. | *You are refactoring the Fe‑S module only* and want to avoid the extra “wrap‑again” step. | Store the =ImplicitSourceTerm= directly (as in the “direct‑term” version).  In that case you can replace =add_implicit_sink= with a tiny variant that updates the term’s =.coeff= attribute instead of a plain number: |

#+begin_src python
def add_implicit_sink_term(LHS, RATES, species, term, rate):
    """Add a sink that is already an ImplicitSourceTerm."""
    if species in LHS:
        LHS[species].coeff += term.coeff   # accumulate the coefficient
    else:
        LHS[species] = term
    RATES[species] = RATES.get(species, 0.0) - float(np.asarray(rate).sum())
#+end_src

#+begin_src python

                                 coeff=-rate_precip * mp["f_diss"])

                       coeff_fe2 * fe2.value)
#+end_src

** 4.  TL;DR checklist

1. *Mathematical equivalence* – both snippets add the same sink term.  
2. *Implementation difference* –  
   * =add_implicit_sink= stores a /number/; later you wrap it in =ImplicitSourceTerm=.  
   * Direct version stores the /=ImplicitSourceTerm=/ itself, so you can use it straight away.  
3. *Pick one style* and be consistent across the whole model.  
   /If you keep the helper, leave it unchanged; if you prefer the direct term, replace the helper with the =*_term= variant shown above./  

Either way the solver will see the identical Jacobian entry, so the numerical results are unchanged.

*** ok, the one problem I see, is that there are other reactions that also affect the same species. So these will add a numerical value. I assume that mixing terms and numerical values will get messy?

*** TL;DR  
/Yes – mixing raw numbers and =ImplicitSourceTerm= objects in the same container quickly becomes confusing./  
The cleanest way is to *store every contribution as a FiPy term* (either a
=ImplicitSourceTerm= or an =ExplicitSourceTerm=) and let FiPy add them together
when the global equation is built.  
If you prefer to keep the old “numeric‑coefficient” style for the slow
reactions, put those numbers in a *separate sub‑dictionary* and convert them
to a term *just before the solve*.  

Below is a small “policy” that works for any number of reactions that share the
same species, together with the minimal changes you have to make to the
existing helpers.

---

** 1.  What “mixed” means in practice  

Assume three reactions touch =fe2_total=  

| Reaction | Contribution |
|----------|--------------|
| Fast precipitation (non‑linear) | =ImplicitSourceTerm(var=fe2, coeff = -rate_precip*f_diss)= |
| Slow oxidation (first order)    | =‑k_ox * fe2=  → a /numeric/ coefficient =‑k_ox= |
| Sorption (linear)               | =‑k_sorp * fe2= → another numeric coefficient =‑k_sorp= |

If you store

#+begin_src python
LHS["fe2_total"] = -k_ox - k_sorp          # plain float
#+end_src

and later also do

#+begin_src python
LHS["fe2_total"] = LHS["fe2_total"] - rate_precip*f_diss   # ← mixing a float with a term
#+end_src

the container now holds /both/ a float *and* a term object.  
When you later write

#+begin_src python
lhs_term = fp.ImplicitSourceTerm(coeff=LHS["fe2_total"], var=fe2)
#+end_src

=coeff= is a /float/, so the precipitation contribution is lost – the matrix
does not contain the Jacobian of the fast reaction.  Conversely, if you keep a
term object and later try to add a float with =LHS["fe2_total"] -= k_ox=,
you will get a type error.

Hence the rule:

> *Never store two different kinds of objects for the same key.*  
> Keep /all/ contributions in the same representation.

---

** 2.  Policy: “everything is a FiPy term”

*** 2.1  Data structures

#+begin_src python
def add_implicit_sink(
    LHS: Dict[str, List[fp.ImplicitSourceTerm]],
    RATES: Dict[str, float],
    species: str,
    coeff: float,
    rate: float,
) -> None:
    """
    Append a *linear* sink ``‑coeff·var`` to ``species``.
    ``coeff`` must be a scalar or a NumPy array of the same shape as the
    variable.  The function creates an ``ImplicitSourceTerm`` and stores it
    in ``LHS[species]``.
    """
    term = fp.ImplicitSourceTerm(var=None, coeff=-coeff)   # var will be filled later
    LHS.setdefault(species, []).append(term)
    # bookkeeping – the raw (non‑linear) rate is only for diagnostics

** 3.  Building the global FiPy equation (one place, one time)

#+begin_src python
def build_and_solve(
    variables: Mapping[str, fp.CellVariable],
    LHS: Dict[str, List[fp.ImplicitSourceTerm]],
    RHS: Dict[str, List[fp.ExplicitSourceTerm]],
    CROSS: Dict[str, List[Tuple[str, float]]],
    dt: float,
) -> None:
    """
    Convert the term‑lists into actual FiPy equations and solve them.
    This function is called **once per time step**.
    """
    # 1️⃣  Attach the real variables to all implicit terms that are still
    #     placeholder (var=None).  This is cheap because we only set the
    #     attribute; the underlying sparse matrix is built later.
    for sp, term_list in LHS.items():
        var = variables[sp]
        for term in term_list:
            if term.var is None:          # only the “sink” helpers left var=None
                term.var = var

    # 2️⃣  Build the coupling terms (target ← source)
    #     For each entry we create a new ImplicitSourceTerm that refers to the
    #     *source* variable.
    for target, couples in CROSS.items():
        target_var = variables[target]
        for source_name, coeff in couples:
            source_var = variables[source_name]
            term = fp.ImplicitSourceTerm(var=target_var,
                                         coeff= coeff * source_var)
            LHS.setdefault(target, []).append(term)

    # 3️⃣  Assemble the full equation for every species and solve.
    for name, var in variables.items():
        # left‑hand side: transient term + all implicit contributions
        lhs = fp.TransientTerm(var=var) + sum(LHS.get(name, []))

        # right‑hand side: explicit sources (may be empty)
        rhs = sum(RHS.get(name, []))

        # FiPy solves: lhs == rhs
        eq = lhs == rhs
        eq.solve(dt=dt)
#+end_src

*** What this achieves

| Feature | How it is handled |
|---------|-------------------|
| *Multiple reactions on the same species* | Each reaction appends its own =ImplicitSourceTerm= (or =ExplicitSourceTerm=) to the list.  No manual coefficient arithmetic is needed. |
| *Fast non‑linear precipitation* | Added as a /single/ =ImplicitSourceTerm= whose =coeff= is the /non‑linear/ FiPy expression =‑rate_precip*f_diss=.  The Jacobian is built automatically. |
| *Slow linear reactions* | Still added via =add_implicit_sink= (which now creates a term object) or via =add_implicit_coupling=. |
| *Cross‑species coupling* | Recorded in =CROSS=; the =build_and_solve= routine turns each entry into a proper term that multiplies the /source/ variable. |
| *Diagnostics* | =RATES= is unchanged – you still get a scalar total rate for each species if you need to log it. |
| *No mixing of types* | =LHS[species]= is /always/ a list of =ImplicitSourceTerm=s; =RHS[species]= is always a list of =ExplicitSourceTerm=s.  The solver never sees a raw float where a term is expected. |

** 5.  Summary checklist

1. *Never store a mix of raw numbers and FiPy term objects* for the same
   species.  
2. Use a *list of =ImplicitSourceTerm=s* (and a separate list of
   =ExplicitSourceTerm=s) as the canonical container.

def _ensure_term_list(
    LHS: Dict[str, Union[float, List[fp.ImplicitSourceTerm]]],
    species: str,
) -> List[fp.ImplicitSourceTerm]:
    """
    Return a list that will hold the implicit terms for *species*.
    If the entry is still a plain float (old style), convert it to a single
    ``ImplicitSourceTerm`` with that coefficient and replace the entry with a
    list containing the term.
    """
    entry = LHS.get(species, None)

    # --------------------------------------------------------------
    # 1️⃣  No entry yet → create an empty list
    # --------------------------------------------------------------
    if entry is None:
        LHS[species] = []                     # type: ignore[assignment]
        return LHS[species]                   # type: ignore[return-value]

    # --------------------------------------------------------------
    # 2️⃣  Entry already a list of terms → just return it
    # --------------------------------------------------------------
    if isinstance(entry, list):
        # we know the list only contains ImplicitSourceTerm objects
        return entry

    # --------------------------------------------------------------
    # 3️⃣  Entry is a plain numeric coefficient → wrap it
    # --------------------------------------------------------------
    if isinstance(entry, (float, int, np.ndarray)):

        # We create a term with that coefficient; the variable will be
        # attached later (see the solver set‑up routine).
        term = fp.ImplicitSourceTerm(var=None, coeff=float(entry))
        LHS[species] = [term]                 # replace the scalar with a list
        return LHS[species]                   # type: ignore[return-value]

    # --------------------------------------------------------------
    # 4️⃣  Anything else is a programming error
    # --------------------------------------------------------------
    raise TypeError(
        f"LHS['{species}'] has unsupported type {type(entry)}. "
        "It must be a float, numpy.ndarray, or a list of ImplicitSourceTerm."
    )

            irr_term = ImplicitSourceTerm(
                coeff=CellVariable(mesh=mesh, value=-D_mol.D_irr), var=var
            ) + CellVariable(mesh=mesh, value=D_mol.D_irr * props["top"])

        # ---- Full equation ----------------------------------------------------
        eq = (TransientTerm(var=var) + conv_term) == (
            diff_term + lhs_term + rhs_term + cross_term + irr_term
        )
        eqs.append(eq)

    coupled_eq = reduce(lambda a, b: a & b, eqs)
    return coupled_eq, species_struct


def update_reaction_rates(
    mp: Any,
    c: Any,
    k: Any,
    diagenetic_reactions: Callable,
    species_struct: List[Dict[str, Any]],
    species_list_partial: List[str],
    current_dt: float,
) -> Tuple[Dict[str, Any], Dict[str, Any]]:
    """
    Compute equilibrium and diagenetic reaction rates, merge them and write the
    new matrix coefficients into the cached FiPy variables.
    """
    from reactions_new import (
        equilibrium_reactions,
    )  # local import to avoid circular deps

    # ---- Equilibrium (operator‑splitting) ------------------------------------
    _, rates_eq = equilibrium_reactions(mp, c, k, None, {}, current_dt)

    # ---- Diagenetic reactions ------------------------------------------------
    f_res, rates = diagenetic_reactions(mp, c, k, f=data_container())

    # Merge equilibrium rates into diagenetic ones (for reporting only)
    for sp in rates:
        if sp in rates_eq:
            rates[sp] += rates_eq[sp]

    # ---- Update LHS / RHS / cross coefficients -------------------------------
    for struct in species_struct:
        name = struct["name"]
        res_tuple = getattr(f_res, name)

        # Diagonal (LHS) and source (RHS)
        struct["lhs"].setValue(getattr(res_tuple[0], "value", res_tuple[0]))
        struct["rhs"].setValue(getattr(res_tuple[1], "value", res_tuple[1]))

        # Cross‑terms, if any
        if len(res_tuple) > 3:
            batch = {src: 0.0 for src in struct["cross"]}
            for src_name, coeff in res_tuple[3]:
                if src_name in batch:
                    batch[src_name] += getattr(coeff, "value", coeff)

            for src, val in batch.items():
                struct["cross"][src].setValue(val)

    return f_res, rates


def adaptive_step(
    coupled_eq: Any,
    solver: Any,
    current_dt: float,
    species_struct: List[Dict[str, Any]],
    mp: Any,
    c: Any,
    k: Any,
    diagenetic_reactions: Callable,
    species_list_partial: List[str],
) -> Tuple[bool, float, Dict[str, np.ndarray]]:
    """
    Perform one pseudo‑transient step.  If the linear solve fails, the time step
    is reduced and the step is retried.  The function returns a tuple::

        (step_converged, new_dt, backup_values)

    where ``backup_values`` is a mapping ``species_name → old_value`` used by
    the caller for convergence diagnostics.
    """
    # Save the current solution (so we can roll back on failure)
    backup = {s["name"]: s["var"].value.copy() for s in species_struct}
    cut_factor = 0.5

    while True:
        try:
            # Update reaction rates *before* solving
            update_reaction_rates(
                mp,
                c,
                k,
                diagenetic_reactions,
                species_struct,
                species_list_partial,
                current_dt,
            )

            # Sweep the coupled system
            
            coupled_eq.sweep(dt=current_dt, solver=solver)
            return True, current_dt, backup
        except Exception as exc:  # noqa: BLE001 (we want to catch FiPy failures)
            print(
                f"  Step failed at dt={current_dt:.1e}: {exc}. "
                f"Retrying with dt*{cut_factor:.2f}"
            )
            traceback.print_exc()
            current_dt *= cut_factor
            if current_dt < 1e-5:
                raise RuntimeError(
                    "Time step became too small – model appears stiff."
                ) from exc

            # Restore previous values before retrying
            for s in species_struct:
                s["var"].value[:] = backup[s["name"]]

def run_non_steady_state_solver_coupled(
    mp: Any,
    c: Any,
    species_list_full: List[str],
    species_list_partial: List[str],
    k: Any,
    diagenetic_reactions: Callable,
    mesh: Any,
    D_mol: Any,
    bc_map: Dict[str, Dict[str, Any]],
    z: Any,
) -> Tuple[int, float]:
    """
    Solve the coupled transport‑reaction system using an adaptive pseudo‑transient
    scheme.

    Parameters
    ----------
    mp
        Model‑parameter container (holds ``solver_backend``, tolerances, etc.).
    c
        Container with FiPy ``CellVariable`` fields for each species.
    species_list_full
        List of *all* species (used only for final data output).
    species_list_partial
        Sub‑set of species that appear in the transport‑reaction system.
    k
        Kinetic‑parameter container passed to the reaction functions.
    diagenetic_reactions
        Callable ``(mp, c, k, f) → (f_res, rates)``.
    mesh
        FiPy mesh on which the variables live.
    D_mol
        Diffusivity container (has attributes for each species plus ``D_bio`` &
        ``D_irr``).
    bc_map
        Boundary‑condition map keyed by species name.
    z
        Depth coordinate array (only used for output).

    Returns
    -------
    steps_done : int
        Number of pseudo‑transient steps performed.
    final_max_change : float
        Largest absolute change in any variable during the last successful step.
    """
    start_wall = time.time()

    # --------------------------------------------------------------------- #
    # 1️⃣ Solver selection
    # --------------------------------------------------------------------- #
    solver = select_solver(mp)

    # --------------------------------------------------------------------- #
    # 2️⃣ Build FiPy variables (LHS/RHS/cross) and equations
    # --------------------------------------------------------------------- #
    lhs_vars, rhs_vars, cross_vars = build_variables(
        mesh, species_list_partial, diagenetic_reactions, mp, c, D_mol
    )
    coupled_eq, species_struct = build_equations(
        mesh,
        species_list_partial,
        bc_map,
        c,
        D_mol,
        mp,
        lhs_vars,
        rhs_vars,
        cross_vars,
    )

    # --------------------------------------------------------------------- #
    # 3️⃣ Adaptive time‑stepping loop
    # --------------------------------------------------------------------- #
    dt_min = getattr(mp, "dt_min", 1e-5 * 365 * 24 * 3600)  # default: ≈ minutes
    current_dt = dt_min
    dt_max = mp.dt_max
    growth_factor = 1.2
    cut_factor = 0.5

    print(
        f"Starting Pseudo‑Transient Solver (Adaptive dt) "
        f"{get_time_units(current_dt):.2f~P}"
    )

    step = 0
    total_time = 0.0
    last_max_change = np.inf
    status = "max steps or time reached"

    try:
        while total_time < mp.t_end and step < mp.max_steps:
            step += 1

            # Advance FiPy variables to the new time level
            for s in species_struct:
                s["var"].updateOld()

            # Keep a copy of the solution before we attempt the step
            backup = {s["name"]: s["var"].value.copy() for s in species_struct}

            # -------------------------------------------------------------- #
            # Adaptive step – may shrink dt and retry internally
            # -------------------------------------------------------------- #
            step_ok, current_dt, _ = adaptive_step(
                coupled_eq,
                solver,
                current_dt,
                species_struct,
                mp,
                c,
                k,
                diagenetic_reactions,
                species_list_partial,
            )
            if not step_ok:
                raise RuntimeError("Adaptive step could not converge.")  # safety

            # -------------------------------------------------------------- #
            # Compute max change (steady‑state criterion)
            # -------------------------------------------------------------- #
            max_change = max(
                np.max(np.abs(s["var"].value - backup[s["name"]]))
                for s in species_struct
            )
            total_time += current_dt

            # -------------------------------------------------------------- #
            # Adapt the time step based on convergence trend
            # -------------------------------------------------------------- #
            if step > 1:
                if max_change < last_max_change:
                    current_dt = min(current_dt * growth_factor, dt_max)
                else:
                    current_dt = max(current_dt * cut_factor, dt_min)
            else:
                # First step – use the original tolerance logic
                if max_change < mp.dt_tolerance:
                    current_dt = min(current_dt * growth_factor, dt_max)

            last_max_change = max_change

            # -------------------------------------------------------------- #
            # Reporting / data output
            # -------------------------------------------------------------- #
            report_progress(
                step,
                mp,
                total_time,
                current_dt,
                max_change,
                c,
                k,
                species_list_full,
                z,
                D_mol,
                diagenetic_reactions,
            )

            # -------------------------------------------------------------- #
            # Steady‑state check
            # -------------------------------------------------------------- #
            if max_change < mp.dt_tolerance:
                print(f"Steady State criteria met (max_change = {max_change:.2e})")
                status = "Converged" if step < mp.max_steps else "Failed"
                break

    except KeyboardInterrupt:
        status = "Interrupted"
    finally:
        # ----------------------------------------------------------------- #
        # Final data write and wall‑time report
        # ----------------------------------------------------------------- #
        save_data_async(
            mp, c, k, species_list_full, z, D_mol, diagenetic_reactions, current_dt
        )
        wall = time.time() - start_wall
        print(f"{status} in {step} steps. Wall time: {wall:.2f}s")

    return step, max_change

*** Why the =float(coeff)= blew up

=coeff= in your fast‑precipitation term is *not a single number* – it is a
=numpy.ndarray= (or a FiPy =CellVariable=) that has one value per mesh cell.  
When you write

#+begin_src python
new_term = fp.ImplicitSourceTerm(var=None, coeff=-float(coeff))
#+end_src

Python tries to convert the whole array to a Python scalar.  That works only
when the array has length 1, otherwise FiPy raises

#+begin_src 
TypeError: only length‑1 arrays can be converted to Python scalars
#+end_src

The fix is simple: *don’t cast the coefficient to =float=*.  FiPy accepts a
scalar, a NumPy array, or a FiPy expression directly as the =coeff=
argument, so we can pass the object unchanged (or, if you really need a scalar,
use =np.asscalar(coeff)= after you have verified it is length‑1).

Below is a minimal, type‑annotated rewrite of the two helper functions that
works for both scalar and array coefficients.  The rest of your code can stay
exactly as you wrote it.

** 1.  Helper that guarantees a homogeneous list of terms

#+begin_src python
from __future__ import annotations

import numpy as np
import fipy as fp
from typing import Dict, List, Tuple, Union

# ----------------------------------------------------------------------
# Internal utility – make sure LHS[species] is a list of ImplicitSourceTerms
# ----------------------------------------------------------------------
def _ensure_term_list(
    LHS: Dict[str, Union[float, np.ndarray, List[fp.ImplicitSourceTerm]]],
    species: str,
) -> List[fp.ImplicitSourceTerm]:
    """
    Return a list that will hold ``ImplicitSourceTerm`` objects for *species*.
    If the entry is still a legacy scalar/array we wrap it in a single term
    and replace the entry with a list.  After the call the dictionary entry
    is **always** a ``list[ImplicitSourceTerm]``.
    """
    entry = LHS.get(species, None)

    # No entry yet → start an empty list
    if entry is None:
        LHS[species] = []                     # type: ignore[assignment]
        return LHS[species]                   # type: ignore[return-value]

    # Already a list → nothing to do
    if isinstance(entry, list):
        return entry

    # Legacy scalar or ndarray → wrap it in a term
    if isinstance(entry, (float, int, np.ndarray)):
        term = fp.ImplicitSourceTerm(var=None, coeff=entry)   # keep the shape
        LHS[species] = [term]                                 # replace
        return LHS[species]                                   # type: ignore[return-value]

    raise TypeError(
        f"LHS['{species}'] has unsupported type {type(entry)} – "
        "must be float, ndarray or list[ImplicitSourceTerm]."
    )
#+end_src

def add_implicit_sink(
    LHS: Dict[str, Union[float, np.ndarray, List[fp.ImplicitSourceTerm]]],
    RATES: Dict[str, float],
    species: str,
    coeff: Union[float, np.ndarray],
    rate: Union[float, np.ndarray],
) -> None:
    """
    Append a linear sink ``‑coeff·var`` to the equation for *species*.

    Parameters
    ----------
    LHS
        Dictionary that may still contain legacy scalar/array entries.
    RATES
        Diagnostic container – the sum of the (non‑linear) rates.
    species
        Name of the variable that receives the sink.
    coeff
        Linear coefficient (positive).  The helper automatically flips the sign.
    rate
        Raw rate value (used only for bookkeeping).
    """

    #     FiPy will accept a scalar, a NumPy array or a CellVariable here.

    RATES[species] = RATES.get(species, 0.0) + float(np.asarray(rate).sum())
